class Tee:
    def __init__(self, path):
        self.f = open(path, "w", encoding="utf-8"); self.s = sys.stdout
    def write(self, x): self.s.write(x); self.f.write(x)
    def flush(self): self.s.flush(); self.f.flush()


def sha(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for c in iter(lambda: f.read(1 << 20), b""):
            h.update(c)
    return h.hexdigest()


def crossings(seq):
    """All zero-crossings of (x, y) pairs by linear interpolation, plus an
    extrapolated crossing past the last interval if y hasn't crossed."""
    xs = []
    for (x0, y0), (x1, y1) in zip(seq, seq[1:]):
        if y0 == 0 or y0 * y1 < 0:
            xs.append((x0 + (x1 - x0) * (0 - y0) / (y1 - y0), "interpolated"))
    if not xs or (seq[-1][1] * seq[-2][1] > 0):
        (x0, y0), (x1, y1) = seq[-2], seq[-1]
        if y1 != y0 and abs(y1) < abs(y0):        # heading toward zero past the edge
            xe = x1 + (x1 - x0) * (0 - y1) / (y1 - y0)
            if xe > x1:
                xs.append((xe, "extrapolated-past-grid-edge"))
    return xs
