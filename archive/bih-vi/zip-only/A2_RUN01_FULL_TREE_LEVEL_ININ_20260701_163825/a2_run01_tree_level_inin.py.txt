# ============================================================
# Numerical conventions
# ============================================================

MPL = 1.0
HC_OVER_MPL = 3.8e-6

# Frozen BIH-VI benchmark
P = 3
MU = 1.0
CS = 1.0
N_STAR = 55
Y_G = 2.0 / 3.0
Z_G = 1.0 / 3.0
DELTA = 0.01
M_POST = 4.0

# Dimensionless normalization: a_g = 1, H_g = 1.
# Shape grid K means perimeter/(3 a_g H_g).
N_MIN = -8.0
N_MAX = 6.0
N_STEPS = 7001

# In-in contour damping for early oscillations.
I_EPS = 0.025


# ============================================================
# Utilities
# ============================================================

def ensure_dir(p: Path) -> Path:
    p.mkdir(parents=True, exist_ok=True)
    return p


def sha256(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def write_json(path: Path, obj: Any) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(obj, indent=2), encoding="utf-8")


def write_csv(path: Path, rows: list[dict[str, Any]]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    if not rows:
        path.write_text("", encoding="utf-8")
        return
    cols = list(rows[0].keys())
    with path.open("w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=cols)
        w.writeheader()
        w.writerows(rows)


def read_csv_dicts(path: Path) -> list[dict[str, Any]]:
    with path.open("r", newline="", encoding="utf-8") as f:
        return list(csv.DictReader(f))


def latest_dir(root: Path, prefix: str) -> Path | None:
    if not root.exists():
        return None
    dirs = sorted([p for p in root.glob(prefix + "*") if p.is_dir()])
    return dirs[-1] if dirs else None


def trapz(y, x):
    if hasattr(np, "trapezoid"):
        return np.trapezoid(y, x)
    return np.trapz(y, x)


def safe_float(x, default=np.nan):
    try:
        y = float(x)
        return y if math.isfinite(y) else default
    except Exception:
        return default

def build_shape_grid() -> list[dict[str, Any]]:
    rows = []
    K_values = list(np.concatenate([
        np.linspace(0.1, 1.0, 10),
        np.linspace(1.25, 3.25, 9),
        np.linspace(3.35, 3.85, 21),
        np.linspace(4.0, 6.0, 9),
    ]))

    for K in K_values:
        rows.append({
            "shape_id": f"equilateral_K{K:.4f}",
            "family": "equilateral",
            "K": float(K),
            "q1": 1.0,
            "q2": 1.0,
            "q3": 1.0,
            "triangle_valid": True
        })

    for q in [1e-3, 1e-2, 1e-1]:
        for K in K_values:
            rows.append({
                "shape_id": f"squeezed_q{q:g}_K{K:.4f}",
                "family": "squeezed",
                "K": float(K),
                "q1": float(q),
                "q2": 1.0,
                "q3": 1.0,
                "triangle_valid": bool(q > 0.0)
            })

    for delta in [1e-2, 1e-3]:
        for K in K_values:
            # k1=k2=a, k3=2a(1-delta). Store ratios to k3.
            q1 = 1.0 / (2.0 * (1.0 - delta))
            q2 = q1
            rows.append({
                "shape_id": f"folded_delta{delta:g}_K{K:.4f}",
                "family": "folded_flattened",
                "K": float(K),
                "q1": float(q1),
                "q2": float(q2),
                "q3": 1.0,
                "triangle_valid": True
            })

    for q in [1e-3, 3e-3, 1e-2, 3e-2, 0.1, 0.2, 0.5, 0.8, 1.0, 1.2, 1.5, 1.8, 1.95]:
        for K in K_values:
            rows.append({
                "shape_id": f"isosceles_q{q:g}_K{K:.4f}",
                "family": "isosceles",
                "K": float(K),
                "q1": float(q),
                "q2": 1.0,
                "q3": 1.0,
                "triangle_valid": bool(q > 0.0 and q < 2.0)
            })

    return rows


def locate_a2_init(c3_work: Path, a2_work: Path) -> Path | None:
    candidates = []
    for base in [c3_work / "_HANDOFF", a2_work / "_HANDOFF"]:
        if base.exists():
            candidates += [p for p in base.glob("A2_INIT_*") if p.is_dir()]
    if not candidates:
        return None
    return sorted(candidates)[-1]


def load_shape_grid(c3_work: Path, a2_work: Path, out: Path) -> tuple[list[dict[str, Any]], dict[str, Any]]:
    init = locate_a2_init(c3_work, a2_work)
    meta = {
        "a2_init_found": bool(init),
        "a2_init_path": str(init) if init else None,
        "shape_grid_source": "generated_fallback"
    }
    if init:
        grid = init / "A2_SHAPE_SCAN_GRID.csv"
        if grid.exists():
            dst = out / "A2_INIT_SHAPE_GRID_USED.csv"
            shutil.copy2(grid, dst)
            meta["shape_grid_source"] = str(grid)
            return read_csv_dicts(grid), meta
    rows = build_shape_grid()
    write_csv(out / "A2_INIT_SHAPE_GRID_GENERATED_FALLBACK.csv", rows)
    return rows, meta

# Frozen BIH-VI background
# ============================================================

def switch_S(y):
    arg = (y - Y_G) / DELTA
    arg = np.clip(arg, -700, 700)
    return 1.0 / (1.0 + np.exp(arg))


def m_of_y(y):
    z = 1.0 - y
    return MU * (z ** P) + M_POST * switch_S(y)


def dy_dN(y):
    return -m_of_y(y) * y


def rk4_step(y, h):
    k1 = dy_dN(y)
    k2 = dy_dN(y + 0.5 * h * k1)
    k3 = dy_dN(y + 0.5 * h * k2)
    k4 = dy_dN(y + h * k3)
    yn = y + (h / 6.0) * (k1 + 2*k2 + 2*k3 + k4)
    return float(np.clip(yn, 1.0e-9, 1.0 - 1.0e-9))


def build_background() -> dict[str, Any]:
    N = np.linspace(N_MIN, N_MAX, N_STEPS)
    i0 = int(np.argmin(np.abs(N)))
    y = np.empty_like(N)
    y[i0] = Y_G

    for i in range(i0, len(N)-1):
        h = N[i+1] - N[i]
        y[i+1] = rk4_step(y[i], h)

    for i in range(i0, 0, -1):
        h = N[i-1] - N[i]
        y[i-1] = rk4_step(y[i], h)

    m = m_of_y(y)
    H_raw = np.sqrt(np.maximum(y * (2.0 - y), 1e-30))
    H_g = math.sqrt(Y_G * (2.0 - Y_G))
    H = H_raw / H_g
    eps = m * (1.0 - y) / (2.0 - y)
    eps = np.maximum(eps, 1e-14)
    ln_eps = np.log(eps)
    eta = np.gradient(ln_eps, N)
    eta_N = np.gradient(eta, N)
    a = np.exp(N)

    return {
        "N": N,
        "a": a,
        "y": y,
        "m": m,
        "H": H,
        "eps": eps,
        "eta": eta,
        "eta_N": eta_N
    }

# Mode model and kernels
# ============================================================

def zeta_mode(k: float, bg: dict[str, Any]) -> tuple[np.ndarray, np.ndarray]:
    """
    Instantaneous Bunch-Davies / slow-variation mode model.

    This is a numerical tree-level kernel pass, not the final MS-mode integration.
    A2_RUN02 should replace this with exact Mukhanov-Sasaki mode functions.
    """
    N = bg["N"]
    a = bg["a"]
    H = bg["H"]
    eps = bg["eps"]

    x = k / np.maximum(a * H, 1e-30)
    amp = H / (2.0 * np.sqrt(eps) * (k ** 1.5) + 1e-300)
    u = amp * (1.0 + 1j * x) * np.exp(-1j * x)

    u_N = np.gradient(u, N)
    return u, u_N


def dot_products(k1, k2, k3):
    # Triangle law: k1·k2 = (k3^2 - k1^2 - k2^2)/2
    d12 = 0.5 * (k3*k3 - k1*k1 - k2*k2)
    d13 = 0.5 * (k2*k2 - k1*k1 - k3*k3)
    d23 = 0.5 * (k1*k1 - k2*k2 - k3*k3)
    return d12, d13, d23


def shape_to_ks(row: dict[str, Any]) -> tuple[float, float, float]:
    K = safe_float(row.get("K"))
    q1 = safe_float(row.get("q1"))
    q2 = safe_float(row.get("q2"))
    q3 = safe_float(row.get("q3"))
    s = q1 + q2 + q3
    scale = 3.0 * K / s
    return scale * q1, scale * q2, scale * q3


def compute_shape(row: dict[str, Any], bg: dict[str, Any]) -> tuple[list[dict[str, Any]], dict[str, Any]]:
    N = bg["N"]
    a = bg["a"]
    H = bg["H"]
    eps = bg["eps"]
    eta_N = bg["eta_N"]

    shape_id = row["shape_id"]
    family = row["family"]
    K = safe_float(row["K"])
    k1, k2, k3 = shape_to_ks(row)

    u1, u1N = zeta_mode(k1, bg)
    u2, u2N = zeta_mode(k2, bg)
    u3, u3N = zeta_mode(k3, bg)

    u10 = u1[-1]
    u20 = u2[-1]
    u30 = u3[-1]
    outside = u10 * u20 * u30

    d12, d13, d23 = dot_products(k1, k2, k3)
    xsum = (k1 + k2 + k3) / np.maximum(a * H, 1e-30)
    regulator = np.exp(-I_EPS * xsum)

    # Conjugate mode products for in-in integrand.
    v1 = np.conjugate(u1)
    v2 = np.conjugate(u2)
    v3 = np.conjugate(u3)
    v1N = np.conjugate(u1N)
    v2N = np.conjugate(u2N)
    v3N = np.conjugate(u3N)

    # O1: zeta dot(zeta)^2
    C1 = (a**3) * (eps**2) * H
    K1 = C1 * (
        v1 * v2N * v3N +
        v2 * v1N * v3N +
        v3 * v1N * v2N
    ) * regulator

    # O2: zeta (grad zeta)^2 / a^2
    # Gradient factors use k_i dot k_j. Overall sign convention is tracked but not used as proof.
    C2 = a * (eps**2) / np.maximum(H, 1e-30)
    K2 = C2 * (
        (-d23) * v1 * v2 * v3 +
        (-d13) * v2 * v1 * v3 +
        (-d12) * v3 * v1 * v2
    ) * regulator

    # O3: dot(zeta) partial_i zeta partial_i chi
    # Constraint kernel included in a conservative schematic form.
    # It must be replaced/audited against the exact chi constraint in A2_RUN02.
    C3 = (a**3) * (eps**2) * H
    invk1 = 1.0 / max(k1*k1, 1e-300)
    invk2 = 1.0 / max(k2*k2, 1e-300)
    invk3 = 1.0 / max(k3*k3, 1e-300)
    K3 = C3 * (
        (d23 * invk1) * v1N * v2 * v3 +
        (d13 * invk2) * v2N * v1 * v3 +
        (d12 * invk3) * v3N * v1 * v2
    ) * regulator

    # O4: epsilon dot(eta) zeta^2 dot(zeta)
    C4 = 0.5 * (a**3) * eps * H * eta_N
    K4 = C4 * (
        v1 * v2 * v3N +
        v1 * v3 * v2N +
        v2 * v3 * v1N
    ) * regulator

    kernels = {
        "O1_zeta_dotzeta2": K1,
        "O2_zeta_gradzeta2": K2,
        "O3_dotzeta_gradzeta_gradchi_SCHEMATIC": K3,
        "O4_switch_eps_doteta_zeta2_dotzeta": K4
    }

    P1 = (k1**3 / (2.0 * math.pi**2)) * abs(u10)**2
    P2 = (k2**3 / (2.0 * math.pi**2)) * abs(u20)**2
    P3 = (k3**3 / (2.0 * math.pi**2)) * abs(u30)**2
    Psum = P1*P2 + P2*P3 + P3*P1
    Psum = max(float(Psum), 1e-300)

    rows = []
    Bsum = 0.0 + 0.0j

    for op, kern in kernels.items():
        integ = trapz(kern, N)
        B = -2.0 * np.imag(outside * integ)
        Bc = complex(float(B), 0.0)
        Bsum += Bc
        fNL = (5.0 / 6.0) * (B / Psum)
        RB = abs(B) / Psum

        status = "ok"
        if "SCHEMATIC" in op:
            status = "requires_exact_constraint_kernel_audit"

        rows.append({
            "shape_id": shape_id,
            "family": family,
            "K": K,
            "k1": k1,
            "k2": k2,
            "k3": k3,
            "operator": op,
            "B_real": float(np.real(Bc)),
            "B_imag": float(np.imag(Bc)),
            "B_abs": float(abs(Bc)),
            "fNL_shape": float(fNL),
            "R_B": float(RB),
            "P1": float(P1),
            "P2": float(P2),
            "P3": float(P3),
            "Psum": float(Psum),
            "mode_model": "instantaneous_BD_slow_variation",
            "kernel_status": status
        })

    # O5 boundary/field redefinition placeholder.
    rows.append({
        "shape_id": shape_id,
        "family": family,
        "K": K,
        "k1": k1,
        "k2": k2,
        "k3": k3,
        "operator": "O5_field_redefinition_boundary_TODO",
        "B_real": 0.0,
        "B_imag": 0.0,
        "B_abs": 0.0,
        "fNL_shape": 0.0,
        "R_B": 0.0,
        "P1": float(P1),
        "P2": float(P2),
        "P3": float(P3),
        "Psum": float(Psum),
        "mode_model": "not_computed",
        "kernel_status": "required_for_final_A2_closure_NOT_COMPUTED_IN_RUN01"
    })

    Bsum_real = float(np.real(Bsum))
    fNL_sum = (5.0 / 6.0) * (Bsum_real / Psum)
    RB_sum = abs(Bsum_real) / Psum

    sumrow = {
        "shape_id": shape_id,
        "family": family,
        "K": K,
        "k1": k1,
        "k2": k2,
        "k3": k3,
        "B_sum_real_O1_to_O4": Bsum_real,
        "B_sum_abs_O1_to_O4": abs(Bsum_real),
        "fNL_sum_O1_to_O4": float(fNL_sum),
        "R_B_sum_O1_to_O4": float(RB_sum),
        "P1": float(P1),
        "P2": float(P2),
        "P3": float(P3),
        "Psum": float(Psum),
        "contains_schematic_O3": True,
        "missing_O5_boundary": True,
        "mode_model": "instantaneous_BD_slow_variation",
        "closure_status": "RUN01_KERNEL_PASS_NOT_FINAL_A2_CLOSURE"
    }

    return rows, sumrow


def classify_summary(sum_rows: list[dict[str, Any]]) -> dict[str, Any]:
    cmb = [r for r in sum_rows if float(r["K"]) < 1.0]
    exit_band = [r for r in sum_rows if 3.35 <= float(r["K"]) <= 3.85]

    def max_abs(rows, key):
        if not rows:
            return None
        return max(rows, key=lambda r: abs(float(r[key])))

    max_cmb_f = max_abs(cmb, "fNL_sum_O1_to_O4")
    max_all_f = max_abs(sum_rows, "fNL_sum_O1_to_O4")
    max_exit_f = max_abs(exit_band, "fNL_sum_O1_to_O4")
    max_absB = max_abs(sum_rows, "B_sum_abs_O1_to_O4")
    max_RB = max_abs(sum_rows, "R_B_sum_O1_to_O4")

    # Strict: Run01 cannot close because O3 is schematic and O5 is missing.
    result = {
        "status": "A2_RUN01_TREE_LEVEL_KERNEL_PASS_COMPLETE",
        "final_A2_closure": False,
        "reason_not_final": [
            "O3 constraint kernel included only as schematic v1; exact chi-constraint kernel audit required.",
            "O5 field-redefinition/boundary contribution is explicitly not computed.",
            "Mode functions use instantaneous BD/slow-variation approximation, not full numerical Mukhanov-Sasaki modes."
        ],
        "allowed_claim": "A2_RUN01 produced an operator-separated tree-level in-in kernel table over the fixed shape grid, suitable for auditing; it is not final A2 closure.",
        "forbidden_claims": [
            "A2 is complete",
            "BIH-VI is proven",
            "exit bispectrum is physically finalized",
            "proxy/canary or Run01 kernel values are observational predictions"
        ],
        "max_cmb_side_abs_fNL_sum": max_cmb_f,
        "max_exit_band_abs_fNL_sum": max_exit_f,
        "max_all_abs_fNL_sum": max_all_f,
        "max_abs_B_sum": max_absB,
        "max_R_B_sum": max_RB
    }

    # Provide a conservative flag for next audit.
    if max_cmb_f is not None and abs(float(max_cmb_f["fNL_sum_O1_to_O4"])) > 10.0:
        result["cmb_safety_flag"] = "REQUIRES_REVIEW_large_K_lt_1_kernel_value"
    else:
        result["cmb_safety_flag"] = "no_large_K_lt_1_flag_in_Run01_kernel_model"

    return result

Allowed claim:

## What this run did

- Loaded the fixed A2 shape grid.
- Built the frozen BIH-VI benchmark background.
- Computed operator-separated numerical in-in kernels for:
  - O1: `zeta dot(zeta)^2`
  - O2: `zeta (partial zeta)^2 / a^2`
  - O3: `dot(zeta) partial_i zeta partial_i chi` (**schematic constraint kernel v1**)
  - O4: `epsilon_H dot(eta_H) zeta^2 dot(zeta)`
- Added explicit O5 placeholder rows so missing boundary/field-redefinition terms are not hidden.
